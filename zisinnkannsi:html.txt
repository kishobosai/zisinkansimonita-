    <script>
        // --- 1. GLOBAL CONFIG & STATE ---
        const CONFIG = {
            usgsUrl: 'https://earthquake.usgs.gov/earthquakes/feed/v1.0/summary/all_day.geojson',
            updateInterval: 60000,
            maxMarkers: 300,
            minMag: 4.0
        };

        let map;
        let earthquakeLayer;
        let knownEventIds = new Set();
        let isSystemActive = false;
        
        let currentMaxMag = 0;
        let priorityEvent = null;

        // Tone.js Instruments
        let synth;
        let drone; // 環境音用

        // --- 2. INITIALIZATION ---
        async function initializeSystem() {
            document.getElementById('start-overlay').style.display = 'none';
            isSystemActive = true;

            await Tone.start();
            initAudio();
            initMap();
            
            fetchData();
            setInterval(fetchData, CONFIG.updateInterval);
            setInterval(updateClock, 1000);
        }

        // --- 3. MAP SETUP ---
        function initMap() {
            map = L.map('map', {
                zoomControl: true, 
                attributionControl: false,
                minZoom: 2, 
                maxBounds: [[-90, -540], [90, 540]], 
                maxBoundsViscosity: 0.5,
            }).setView([36, 140], 5); 

            map.zoomControl.setPosition('bottomright');

            L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', {
                attribution: '&copy; OpenStreetMap &copy; CARTO',
                subdomains: 'abcd',
                maxZoom: 19,
                noWrap: false
            }).addTo(map);

            earthquakeLayer = L.layerGroup().addTo(map);
            L.control.attribution({position: 'bottomright'}).addTo(map);
        }

        // --- 4. DATA FETCHING ---
        async function fetchData() {
            try {
                const response = await fetch(CONFIG.usgsUrl);
                const data = await response.json();
                
                processEarthquakes(data.features);
                updateStats(data.features);
                updateCameraFocus(data.features);

            } catch (error) {
                console.error("Data Fetch Error:", error);
            }
        }

        function updateStats(features) {
            const count = features.length;
            const maxMag = Math.max(...features.map(f => f.properties.mag));
            currentMaxMag = maxMag;

            document.getElementById('count-display').innerText = count;
            document.getElementById('max-mag-display').innerText = maxMag.toFixed(1);
            
            updateAlertStatus(maxMag);
        }

        function updateAlertStatus(maxMag) {
            const statusEl = document.getElementById('status-display');
            const alertBar = document.getElementById('alert-bar');

            if(maxMag >= 6.0) {
                statusEl.innerText = "DANGER";
                statusEl.className = "text-lg font-bold text-red-500 blink leading-none";
                
                alertBar.style.display = 'block';
                const warningText = maxMag >= 7.0 ? "EMERGENCY: M7+ CATASTROPHIC EVENT" : "WARNING: M6+ STRONG QUAKE DETECTED";
                alertBar.innerText = `⚠ ${warningText} ⚠`;
                
            } else if(maxMag >= 4.5) {
                statusEl.innerText = "CAUTION";
                statusEl.className = "text-lg font-bold text-yellow-400 leading-none";
                alertBar.style.display = 'none';
            } else {
                statusEl.innerText = "NORMAL";
                statusEl.className = "text-lg font-bold text-green-400 leading-none";
                alertBar.style.display = 'none';
            }
        }

        // --- 5. CAMERA LOGIC ---
        function updateCameraFocus(features) {
            if (features.length === 0) return;

            const sorted = [...features].sort((a, b) => a.properties.time - b.properties.time);
            const filtered = sorted.filter(f => f.properties.mag >= CONFIG.minMag);
            if (filtered.length === 0) return;
            
            const latest = filtered[filtered.length - 1];
            const hugeEvent = sorted.find(f => f.properties.mag >= 7.0);

            let target = latest;
            if (hugeEvent) {
                target = hugeEvent;
                priorityEvent = hugeEvent;
            } else {
                priorityEvent = null;
            }

            const coords = target.geometry.coordinates;
            const zoom = target.properties.mag >= 6.0 ? 6 : 5;

            map.flyTo([coords[1], coords[0]], zoom, {
                duration: 2.0,
                easeLinearity: 0.25
            });
        }

        // --- 6. PLOTTING & STYLING ---
        function processEarthquakes(features) {
            features.sort((a, b) => a.properties.time - b.properties.time);

            if(knownEventIds.size > CONFIG.maxMarkers) {
                earthquakeLayer.clearLayers();
                knownEventIds.clear();
                document.getElementById('event-list').innerHTML = '';
            }

            features.forEach(feature => {
                const props = feature.properties;
                const coords = feature.geometry.coordinates;
                const id = feature.id;

                if (props.mag < CONFIG.minMag) return;

                if (!knownEventIds.has(id)) {
                    knownEventIds.add(id);
                    plotQuake(coords[1], coords[0], props.mag, props.place, props.time);
                    plotQuake(coords[1], coords[0] - 360, props.mag, props.place, props.time);
                    plotQuake(coords[1], coords[0] + 360, props.mag, props.place, props.time);
                    
                    if (props.mag >= 4.0) {
                        updateEventList(feature);
                    }

                    const now = new Date().getTime();
                    if (now - props.time < 3600000) {
                        playQuakeSound(props.mag);
                    }
                }
            });
        }

        function plotQuake(lat, lng, mag, place, time) {
            let color = '#00f3ff';
            let fillColor = '#00f3ff';
            let radius = mag * 1.5;
            let className = '';
            let opacity = 0.4;

            if (mag < 4.0) {
                color = '#00f3ff'; radius = 2; 
            } else if (mag < 4.5) {
                color = '#00ff9d'; radius = mag * 1.5;
            } else if (mag < 6.0) {
                color = '#ffae00'; fillColor = '#ffae00'; radius = mag * 2.5; opacity = 0.6;
            } else if (mag < 7.0) {
                color = '#ff4000'; fillColor = '#ff4000'; radius = mag * 4; opacity = 0.7; className = 'quake-pulse';
            } else {
                color = '#ff0000'; fillColor = '#ff0055'; radius = mag * 6; opacity = 0.8; className = 'quake-pulse';
            }

            const marker = L.circleMarker([lat, lng], {
                radius: radius, color: color, fillColor: fillColor, fillOpacity: opacity, weight: 1, className: className
            });

            const date = new Date(time).toLocaleTimeString('ja-JP');
            marker.bindPopup(`
                <div style="font-family: 'Noto Sans JP'; color: #000; font-size: 11px; line-height: 1.2;">
                    <strong style="font-size: 13px;">M${mag.toFixed(1)}</strong><br>
                    ${translatePlace(place)}<br>
                    ${date}
                </div>
            `);
            marker.addTo(earthquakeLayer);
        }

        function updateEventList(feature) {
            const list = document.getElementById('event-list');
            const item = document.createElement('div');
            const props = feature.properties;
            
            let colorClass = 'text-cyan-300';
            if (props.mag >= 7.0) colorClass = 'text-red-500 font-black';
            else if (props.mag >= 6.0) colorClass = 'text-red-400 font-bold';
            else if (props.mag >= 4.5) colorClass = 'text-yellow-400';
            
            item.className = `p-1.5 border-b border-gray-800 hover:bg-white/5 transition text-[10px]`;
            item.innerHTML = `
                <div class="flex justify-between items-center">
                    <span class="${colorClass}">M${props.mag.toFixed(1)}</span>
                    <span class="text-gray-400">${new Date(props.time).toLocaleTimeString('ja-JP')}</span>
                </div>
                <div class="truncate text-gray-300 w-full overflow-hidden">${translatePlace(props.place)}</div>
            `;
            
            list.insertBefore(item, list.firstChild);
            if(list.children.length > 10) list.removeChild(list.lastChild);
        }

        // --- 7. AUDIO ---
        function initAudio() {
            // Main Synth
            synth = new Tone.PolySynth(Tone.Synth, {
                oscillator: { type: "triangle" },
                envelope: { attack: 0.01, decay: 0.3, sustain: 0.1, release: 1 }
            }).toDestination();
            const reverb = new Tone.Reverb(3).toDestination();
            synth.connect(reverb);

            // Ambient Drone (Restored)
            drone = new Tone.Noise("brown").start();
            const filter = new Tone.AutoFilter({
                frequency: "8m", min: 800, max: 15000
            }).connect(Tone.Master);
            drone.connect(filter);
            drone.volume.value = -35; // Quiet background rumble
        }

        function playQuakeSound(mag) {
            if (!isSystemActive) return;

            let notes = [];
            if (mag < 4.0) notes = ["C5"];
            else if (mag < 5.0) notes = ["C4", "G4"];
            else if (mag < 6.0) notes = ["C3", "G3", "C4"];
            else if (mag < 7.0) notes = ["C2", "G2", "C3", "D#3"]; 
            else notes = ["C1", "C2", "F#2", "A#2"];

            const velocity = Math.min(mag / 8, 1);
            synth.triggerAttackRelease(notes, "8n", undefined, velocity);
        }

        // --- UTILS ---
        function translatePlace(place) {
            if(!place) return "不明";
            return place
                .replace("[SIMULATION]", "【訓練/テスト】")
                .replace("Japan", "日本")
                .replace("near the coast of", "沿岸")
                .replace("region", "地域")
                .replace("Islands", "諸島")
                .replace("Island", "島")
                .replace("Southern", "南")
                .replace("Northern", "北")
                .replace("Eastern", "東")
                .replace("Western", "西")
                .replace("South of", "南:")
                .replace("North of", "北:")
                .replace("East of", "東:")
                .replace("West of", "西:")
                .replace("km", "km");
        }

        function updateClock() {
            const now = new Date();
            document.getElementById('clock').innerText = 
                now.toLocaleTimeString('ja-JP') + " JST";
        }
